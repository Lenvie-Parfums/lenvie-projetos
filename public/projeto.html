const projetoId = new URLSearchParams(window.location.search).get('id'); const form = document.getElementById('form'); const titulo = document.getElementById('titulo'); const resumoProjeto = document.getElementById('resumoProjeto'); const selectStatus = document.getElementById('status'); const selectEtapa = document.getElementById('etapa'); const selectArea = document.getElementById('area'); const prazo90 = document.getElementById('prazo90'); const temposEtapa = document.getElementById('temposEtapa'); const temposEspera = document.getElementById('temposEspera'); const historico = document.getElementById('historico'); function preencherSelect(elemento, itens = []) { elemento.innerHTML = ''; itens.forEach(item => { const option = document.createElement('option'); option.value = item; option.textContent = item; elemento.appendChild(option); }); } function formatarDataInput(valor) { if (!valor) return ''; return String(valor).slice(0, 10); } function renderizarTempos(dados = [], elemento, chave) { if (!dados.length) { elemento.innerHTML = 'Sem dados ainda.'; return; } elemento.innerHTML = dados.map(item => `
${item[chave] || '—'} ${item.dias ?? 0} d
`).join(''); } function garantirOpcao(select, valor) { if (!valor) return; const existe = Array.from(select.options) .some(option => option.value === valor); if (!existe) { const option = document.createElement('option'); option.value = valor; option.textContent = valor; select.appendChild(option); } select.value = valor; } async function carregarProjeto() { try { if (!projetoId) { throw new Error('ID do projeto não informado.'); } const [configResponse, projetoResponse] = await Promise.all([ fetch('/api/config'), fetch(`/api/projetos/${projetoId}`) ]); if (!configResponse.ok) { throw new Error('Erro ao carregar configurações.'); } if (!projetoResponse.ok) { throw new Error('Erro ao carregar projeto.'); } const config = await configResponse.json(); const dados = await projetoResponse.json(); const projeto = dados.projeto; // ========================================================= // SELECTS // ========================================================= preencherSelect( selectStatus, config.status || [] ); preencherSelect( selectEtapa, config.etapas || [] ); preencherSelect( selectArea, config.areas || [] ); // ========================================================= // CABEÇALHO // ========================================================= titulo.textContent = `${projeto.codigo} · ${projeto.nome}`; const diasEmAberto = projeto.dias_em_aberto ?? 0; let textoPrazo90 = ''; if ( projeto.dias_para_90 !== null && projeto.dias_para_90 !== undefined ) { if (projeto.dias_para_90 >= 0) { textoPrazo90 = ` • ${projeto.dias_para_90} dias para o prazo de 90 dias `; } else { textoPrazo90 = ` • ${Math.abs(projeto.dias_para_90)} dias além do prazo de 90 dias `; } } resumoProjeto.innerHTML = ` ${projeto.situacao_automatica || 'SEM CLASSIFICAÇÃO'} ${diasEmAberto} dias desde o início ${textoPrazo90} `; // ========================================================= // CAMPOS DO FORMULÁRIO // ========================================================= const camposData = [ 'data_inicio', 'previsao_conclusao', 'prazo_proxima_acao', 'data_aprovacao', 'data_conclusao' ]; for ( const [campo, valor] of Object.entries(projeto) ) { // Esses campos serão tratados depois. if ( campo === 'status' || campo === 'etapa_atual' || campo === 'area_pendente' ) { continue; } const elemento = form.elements[campo]; if (!elemento) { continue; } if (camposData.includes(campo)) { elemento.value = formatarDataInput(valor); } else { elemento.value = valor ?? ''; } } // ========================================================= // VALORES DOS SELECTS // ========================================================= garantirOpcao( selectStatus, projeto.status || 'Em andamento' ); garantirOpcao( selectEtapa, projeto.etapa_atual || config.etapas?.[0] || 'Entrada / Oportunidade' ); garantirOpcao( selectArea, projeto.area_pendente || 'Sem pendência' ); // ========================================================= // PRAZO 90 DIAS // ========================================================= prazo90.value = formatarDataInput( projeto.prazo_90_dias ); // ========================================================= // TEMPO POR ETAPA // ========================================================= renderizarTempos( dados.tempos_etapa || [], temposEtapa, 'etapa' ); // ========================================================= // TEMPO AGUARDANDO // ========================================================= renderizarTempos( dados.tempos_espera || [], temposEspera, 'area_pendente' ); // ========================================================= // HISTÓRICO // ========================================================= const registros = dados.historico || []; if (!registros.length) { historico.innerHTML = 'Sem histórico.'; } else { historico.innerHTML = registros.map(item => { let dataRegistro = '—'; if (item.data_registro) { dataRegistro = new Date( item.data_registro ).toLocaleString('pt-BR'); } const dias = Number( item.dias_na_situacao || 0 ); return `
${dataRegistro} • ${dias.toFixed(1)} dias
${item.situacao || ''} ${item.etapa || ''}
Aguardando: ${item.area_pendente || '—'} ${ item.pendencia_proximo_passo ? `
${item.pendencia_proximo_passo}` : '' } ${ item.observacoes ? `
${item.observacoes}` : '' }
`; }).join(''); } } catch (erro) { console.error(erro); alert( 'Não foi possível carregar o projeto.' ); } } // ============================================================= // SALVAR // ============================================================= form.addEventListener( 'submit', async event => { event.preventDefault(); try { const dadosFormulario = Object.fromEntries( new FormData(form).entries() ); // Proteções contra campos críticos vazios. if (!dadosFormulario.status) { dadosFormulario.status = selectStatus.value || 'Em andamento'; } if (!dadosFormulario.etapa_atual) { dadosFormulario.etapa_atual = selectEtapa.value || 'Entrada / Oportunidade'; } if (!dadosFormulario.area_pendente) { dadosFormulario.area_pendente = selectArea.value || 'Sem pendência'; } const response = await fetch( `/api/projetos/${projetoId}`, { method: 'PUT', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify( dadosFormulario ) } ); const resultado = await response.json(); if (!response.ok) { console.error(resultado); alert( resultado.detalhe || resultado.erro || 'Erro ao atualizar projeto.' ); return; } window.location.reload(); } catch (erro) { console.error(erro); alert( 'Erro ao salvar atualização.' ); } } ); // ============================================================= // INICIAR // ============================================================= carregarProjeto();
